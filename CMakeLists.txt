cmake_minimum_required(VERSION 3.20)
project(photo_enhancement VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PE_BUILD_TESTS "Build native tests" ON)
option(PE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(PE_WITH_HEIF "Enable optional HEIF codec" ON)
option(PE_WITH_WEBP "Enable optional WebP decoding" ON)
option(PE_PORTABLE_DEPS "Build pinned dependencies from source" OFF)
option(PE_BUILD_CLI "Build command-line application" ON)
option(PE_STATIC_CORE "Build a static core for iOS linking" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/LibRaw/libraw/libraw.h")
  message(FATAL_ERROR "LibRaw submodule missing. Run: git submodule update --init --recursive")
endif()

if(WIN32)
  include(cmake/WindowsDependencies.cmake)
elseif(PE_PORTABLE_DEPS)
  include(cmake/PortableDependencies.cmake)
else()
if(APPLE AND NOT CMAKE_CROSSCOMPILING)
  list(APPEND CMAKE_PREFIX_PATH /opt/homebrew /opt/homebrew/opt/jpeg-turbo /usr/local)
endif()
find_package(JPEG REQUIRED)
find_package(PNG REQUIRED)
find_package(TIFF 4.6.1 REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Freetype REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LCMS2 REQUIRED IMPORTED_TARGET lcms2)
if(PE_WITH_HEIF)
  pkg_check_modules(HEIF IMPORTED_TARGET libheif>=1.19.7)
endif()
if(PE_WITH_WEBP)
  pkg_check_modules(WEBP IMPORTED_TARGET libwebp)
  pkg_check_modules(WEBPDEMUX IMPORTED_TARGET libwebpdemux)
endif()
endif()

# Build the checked-in source, not the architecture-specific prebuilt archives.
file(GLOB_RECURSE LIBRAW_SOURCES CONFIGURE_DEPENDS third_party/LibRaw/src/*.cpp)
# *_ph.cpp are alternative amalgamated implementations, not extra features.
list(FILTER LIBRAW_SOURCES EXCLUDE REGEX "_ph\\.cpp$")
add_library(pe_raw STATIC ${LIBRAW_SOURCES})
set_target_properties(pe_raw PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_include_directories(pe_raw PUBLIC third_party/LibRaw)
target_compile_definitions(pe_raw PRIVATE USE_ZLIB USE_JPEG USE_LCMS2)
if(WIN32)
  target_compile_definitions(pe_raw PUBLIC LIBRAW_NODLL NOMINMAX)
endif()
target_link_libraries(pe_raw PRIVATE ZLIB::ZLIB JPEG::JPEG PkgConfig::LCMS2)
target_compile_options(pe_raw PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-w>)

if(PE_STATIC_CORE)
  set(PE_CORE_TYPE STATIC)
else()
  set(PE_CORE_TYPE SHARED)
endif()
add_library(photo_core ${PE_CORE_TYPE}
  native/src/api.cpp native/src/color.cpp native/src/lut.cpp
  native/src/border.cpp native/src/frame_template.cpp native/src/codecs.cpp native/src/metadata.cpp native/src/effects.cpp
  native/src/platform_io.cpp native/src/resources.cpp native/src/export.cpp native/src/look.cpp native/src/tiff.cpp)
target_include_directories(photo_core PUBLIC native/include PRIVATE native/src)
target_link_libraries(photo_core PRIVATE pe_raw JPEG::JPEG PNG::PNG
  Freetype::Freetype PkgConfig::LCMS2 TIFF::TIFF)
target_compile_definitions(photo_core PRIVATE PE_BUILDING_LIBRARY)
if(MSVC)
  target_compile_options(pe_raw PRIVATE /utf-8)
  target_compile_options(photo_core PRIVATE /utf-8)
  target_compile_definitions(photo_core PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
set_target_properties(photo_core PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON MACOSX_RPATH ON)
if(PE_WITH_HEIF AND HEIF_FOUND)
  target_link_libraries(photo_core PRIVATE PkgConfig::HEIF)
  target_compile_definitions(photo_core PRIVATE PE_HAVE_HEIF)
endif()
if(PE_WITH_WEBP AND WEBP_FOUND AND WEBPDEMUX_FOUND)
  target_link_libraries(photo_core PRIVATE PkgConfig::WEBP PkgConfig::WEBPDEMUX)
  target_compile_definitions(photo_core PRIVATE PE_HAVE_WEBP)
endif()
target_compile_options(photo_core PRIVATE $<$<CXX_COMPILER_ID:Clang,AppleClang,GNU>:-Wall;-Wextra;-Wpedantic>)

# Resolve archive filenames from targets, including upstream output-name and
# generator/configuration differences. iOS needs this entire static closure.
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" AND PE_STATIC_CORE AND PE_PORTABLE_DEPS)
  set(pe_ios_archives photo_core pe_raw JPEG::JPEG PNG::PNG TIFF::TIFF
    ZLIB::ZLIB pe_lcms2 Freetype::Freetype)
  if(PE_WITH_WEBP)
    list(APPEND pe_ios_archives webp webpdemux sharpyuv)
  endif()
  if(PE_WITH_HEIF)
    list(APPEND pe_ios_archives heif de265)
  endif()
  set(pe_ios_archive_paths "")
  foreach(archive IN LISTS pe_ios_archives)
    string(APPEND pe_ios_archive_paths "$<TARGET_FILE:${archive}>\n")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/ios-core-archives-$<CONFIG>.txt"
    CONTENT "${pe_ios_archive_paths}")
endif()

if(PE_SANITIZE)
  foreach(target pe_raw photo_core)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()

if(PE_BUILD_CLI)
  add_executable(photo_cli native/cli/main.cpp)
  target_link_libraries(photo_cli PRIVATE photo_core)
  install(TARGETS photo_cli RUNTIME DESTINATION bin)
endif()
if(PE_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(photo_platform_io_tests native/tests/platform_io_tests.cpp native/src/platform_io.cpp)
  target_include_directories(photo_platform_io_tests PRIVATE native/src)
  target_link_libraries(photo_platform_io_tests PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(photo_platform_io_tests PRIVATE /utf-8)
  endif()
  add_test(NAME photo_platform_io_tests COMMAND photo_platform_io_tests)
  # Legacy native fixtures use POSIX test helpers. The portable I/O suite runs
  # on both Windows and Unix; keep legacy fixtures on their supported hosts.
  if(NOT WIN32)
    add_executable(photo_core_tests native/tests/core_tests.cpp)
    target_link_libraries(photo_core_tests PRIVATE photo_core JPEG::JPEG PNG::PNG PkgConfig::LCMS2)
    target_compile_definitions(photo_core_tests PRIVATE PE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_include_directories(photo_core_tests PRIVATE native/src)
    target_compile_definitions(photo_core_tests PRIVATE PE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME photo_core_tests COMMAND photo_core_tests)
    add_executable(photo_tiff_tests native/tests/tiff_tests.cpp)
    target_link_libraries(photo_tiff_tests PRIVATE photo_core TIFF::TIFF PNG::PNG PkgConfig::LCMS2 Threads::Threads)
    add_test(NAME photo_tiff_tests COMMAND photo_tiff_tests)
    add_executable(photo_raw_export_tests native/tests/raw_export_tests.cpp)
    target_link_libraries(photo_raw_export_tests PRIVATE photo_core TIFF::TIFF)
    target_compile_definitions(photo_raw_export_tests PRIVATE PE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME photo_raw_export_tests COMMAND photo_raw_export_tests)
  endif()
endif()
install(TARGETS photo_core LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY native/include/ DESTINATION include)
